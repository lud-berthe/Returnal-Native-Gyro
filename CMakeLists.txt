cmake_minimum_required(VERSION 3.24)
project(ReturnalGyro VERSION 1.0.1 LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with MSVC for Windows x64.")
endif()
find_package(SDL3 CONFIG REQUIRED PATHS "${CMAKE_CURRENT_SOURCE_DIR}/_deps/SDL3-3.4.16/cmake")
set(MH third_party/minhook-1.3.4)
add_library(minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(minhook PUBLIC ${MH}/include PRIVATE ${MH}/src ${MH}/src/hde)
add_library(GyroCore STATIC src/settings.cpp src/motion.cpp src/gyro_menu.cpp src/localization.cpp)
target_include_directories(GyroCore PUBLIC include)
target_include_directories(GyroCore SYSTEM PUBLIC third_party)
target_compile_options(GyroCore PUBLIC /utf-8)
target_compile_definitions(GyroCore PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(MotionBackend STATIC src/sony_backend.cpp src/sdl_backend.cpp src/steam_backend.cpp src/auto_backend.cpp)
target_link_libraries(MotionBackend PUBLIC GyroCore SDL3::SDL3)
add_library(ReturnalGyro SHARED src/runtime.cpp src/runtime_state.cpp src/bindings.cpp src/ipc.cpp src/control_server.cpp src/native_menu.cpp)
target_link_libraries(ReturnalGyro PRIVATE MotionBackend minhook bcrypt)
add_custom_command(TARGET ReturnalGyro POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:ReturnalGyro>)
add_library(ReturnalLoader SHARED src/version_proxy.cpp src/version_stubs.asm src/version.def)
target_include_directories(ReturnalLoader PRIVATE include)
set_target_properties(ReturnalLoader PROPERTIES OUTPUT_NAME version RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/loader")
foreach(target GyroCore MotionBackend ReturnalGyro ReturnalLoader)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/EHsc>)
endforeach()
enable_testing()
function(gyro_test target name)
  add_executable(${target} ${ARGN})
  target_link_libraries(${target} PRIVATE GyroCore)
  add_test(NAME ${name} COMMAND ${target})
endfunction()
gyro_test(GyroTests GyroCore tests/core_tests.cpp)
gyro_test(PolicyTests InputPolicyAndDpi tests/policy_tests.cpp)
gyro_test(SettingsIpcTests SettingsIPC tests/ipc_tests.cpp src/ipc.cpp)
gyro_test(MenuTests IntegratedMenuModel tests/menu_tests.cpp)
gyro_test(ControllerTests ControllerButtons tests/controller_tests.cpp)
gyro_test(FlickTests FlickStick tests/flick_tests.cpp)
gyro_test(ShortPressTests ShortPressAndCalibration tests/short_press_tests.cpp)
gyro_test(HapticGateTests MixedInputHaptics tests/haptic_gate_tests.cpp)
gyro_test(SteamMotionTests SteamMotion tests/steam_tests.cpp)
gyro_test(AimGateTests GyroAimCommand tests/aim_gate_tests.cpp)
gyro_test(SteamContactTests SteamContacts tests/steam_contact_tests.cpp)
gyro_test(BindingFamilyTests GyroBindingFamilies tests/binding_family_tests.cpp)
gyro_test(SteamAssociationTests SteamAssociation tests/steam_association_tests.cpp)
gyro_test(OrientationTests MotionOrientationRecovery tests/orientation_tests.cpp)
add_executable(VersionProxyTest tools/version_proxy_test.cpp)
add_test(NAME VersionProxy COMMAND VersionProxyTest $<TARGET_FILE:ReturnalLoader>)
